### Task 1: `/api/chat` 代理路由

**Files:**
- Create: `web/src/app/api/chat/route.ts`

**Interfaces:**
- Produces: `POST /api/chat`，body `{text: string, sessionId: string}`
  - 成功：`200` SSE 流（`text/event-stream`，透传 QwenPaw 事件）
  - 未配 env：`503 {error:'agent-unreachable'}`
  - 上游不可达/非 2xx：`502 {error:'agent-error'}`
  - 上游超时（180s 无响应头）：`504 {error:'agent-timeout'}`
  - 超限：`429 {error:'rate-limited'}`；参数错：`400 {error:'bad-request'}`

- [ ] **Step 1: 写实现**（无单测框架，验证=Step 2 curl 行为测试）

```ts
/* web/src/app/api/chat/route.ts —— QwenPaw 代理：SSE 透传；密钥只经 env（密钥纪律） */
export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

const RATE_MAX = 20, RATE_WIN = 60_000, MAX_LEN = 500, TIMEOUT_MS = 180_000;
const hits = new Map<string, number[]>(); /* 实例内内存限流，演示级足够 */

function limited(ip: string): boolean {
  const now = Date.now(), a = (hits.get(ip) ?? []).filter(t => now - t < RATE_WIN);
  a.push(now); hits.set(ip, a);
  return a.length > RATE_MAX;
}

export async function POST(req: Request) {
  const ip = req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || 'local';
  let body: { text?: unknown; sessionId?: unknown };
  try { body = await req.json(); } catch { return Response.json({ error: 'bad-request' }, { status: 400 }); }
  const text = typeof body.text === 'string' ? body.text.trim() : '';
  const sessionId = typeof body.sessionId === 'string' ? body.sessionId.slice(0, 64) : '';
  if (!text || text.length > MAX_LEN || !sessionId) return Response.json({ error: 'bad-request' }, { status: 400 });
  if (limited(ip)) return Response.json({ error: 'rate-limited' }, { status: 429 });

  const base = process.env.QWENPAW_URL;
  if (!base) return Response.json({ error: 'agent-unreachable' }, { status: 503 });

  const ac = new AbortController();
  const timer = setTimeout(() => ac.abort(), TIMEOUT_MS);
  try {
    const up = await fetch(`${base.replace(/\/$/, '')}/api/console/chat`, {
      method: 'POST',
      signal: ac.signal,
      headers: {
        'Content-Type': 'application/json',
        'X-Agent-Id': process.env.QWENPAW_AGENT_ID || 'default',
        ...(process.env.QWENPAW_TOKEN ? { Authorization: `Bearer ${process.env.QWENPAW_TOKEN}` } : {}),
      },
      body: JSON.stringify({
        input: [{ role: 'user', content: [{ type: 'text', text }] }],
        session_id: sessionId, user_id: 'zhulong-web', channel: 'console',
      }),
    });
    if (!up.ok || !up.body) return Response.json({ error: 'agent-error' }, { status: 502 });
    return new Response(up.body, {
      headers: {
        'Content-Type': 'text/event-stream; charset=utf-8',
        'Cache-Control': 'no-store, no-transform',
        'X-Accel-Buffering': 'no',
      },
    });
  } catch (e) {
    const timedOut = e instanceof Error && e.name === 'AbortError';
    return Response.json({ error: timedOut ? 'agent-timeout' : 'agent-error' }, { status: timedOut ? 504 : 502 });
  } finally { clearTimeout(timer); }
}
```

- [ ] **Step 1: 写实现**（快照/增量自适应：QwenPaw 事件 `output` 可能是累计快照或增量，两者都兼容）

```ts
/* web/src/lib/zl/chat.ts —— /api/chat SSE 消费：快照/增量自适应渲染，纯 DOM 回调（engine 命令式模式） */
export interface ChatCbs { onText(full: string): void; onStatus(s: 'thinking' | 'done' | 'error'): void }

export async function streamChat(text: string, sessionId: string, cbs: ChatCbs): Promise<{ ok: boolean; error?: string }> {
  let res: Response;
  try {
    res = await fetch('/api/chat', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text, sessionId }),
    });
  } catch { cbs.onStatus('error'); return { ok: false, error: 'network' }; }
  if (!res.ok || !res.body) {
    let code = `http-${res.status}`;
    try { code = (await res.json()).error || code; } catch { /* keep http code */ }
    cbs.onStatus('error'); return { ok: false, error: code };
  }
  cbs.onStatus('thinking');
  const reader = res.body.getReader(), dec = new TextDecoder();
  let buf = '', acc = ''; /* acc=已积累文本；事件文本若是快照（startsWith acc）则替换，否则追加 */
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    const lines = buf.split('\n'); buf = lines.pop() ?? '';
    for (const line of lines) {
      if (!line.startsWith('data: ')) continue;
      let ev: {
        status?: string; error?: { message?: string };
        output?: { role?: string; content?: { type?: string; text?: string }[] }[];
      };
      try { ev = JSON.parse(line.slice(6)); } catch { continue; }
      if (ev.error) { cbs.onStatus('error'); return { ok: false, error: ev.error.message || 'agent-error' }; }
      const t = (ev.output ?? []).filter(o => o.role === 'assistant')
        .flatMap(o => o.content ?? []).filter(c => c.type === 'text').map(c => c.text ?? '').join('');
      if (t) { acc = t.startsWith(acc) ? t : acc + t; cbs.onText(acc); }
      if (ev.status === 'completed') { cbs.onStatus('done'); return { ok: true }; }
      if (ev.status === 'failed') { cbs.onStatus('error'); return { ok: false, error: 'agent-failed' }; }
    }
  }
  cbs.onStatus(acc ? 'done' : 'error');
  return { ok: !!acc, error: acc ? undefined : 'stream-end' };
}
```

/* ===== feat-023 ChatBI 数据问答（互斥家族第五件） ===== */
import { streamChat } from './chat';
import { state } from './store'; // 上下文注入只读 state（zone/mode/origin）

function mountChat() {
  let sessionId = '';
  const layer = () => $('chatLayer');
  const openChat = () => {
    (['calPopover', 'optPopover', 'basisPopover', 'sqTip'] as const).forEach(id => $(id).style.display = 'none');
    if (!sessionId) sessionId = crypto.randomUUID();           /* 每次面板生命周期一个会话 */
    layer().classList.add('on');
    $('chatInput').focus();
  };
  const closeChat = () => layer().classList.remove('on');

  const send = async (raw: string) => {
    const text = raw.trim(); if (!text) return;
    const log = $('chatLog');
    const ctx = `[页面上下文] 区域=${state.zone}; 视图=${state.mode === 'live' ? '实时' : '重演@' + new Date(state.origin).toISOString().slice(0, 10)}\n`;
    log.insertAdjacentHTML('beforeend',
      `<div class="chat-msg user"></div>`);
    (log.lastElementChild as HTMLElement).textContent = text;
    const typing = document.createElement('div');
    typing.className = 'chat-typing'; typing.innerHTML = '<span class="spin"></span>正在查询数据、执行分析…';
    log.appendChild(typing);
    const bubble = document.createElement('div'); bubble.className = 'chat-msg bot'; bubble.style.display = 'none';
    log.appendChild(bubble);
    log.scrollTop = log.scrollHeight;
    ($('chatSend') as HTMLButtonElement).disabled = true; ($('chatInput') as HTMLInputElement).value = '';

    const r = await streamChat(ctx + text, sessionId, {
      onText: full => { typing.remove(); bubble.style.display = ''; bubble.textContent = full; log.scrollTop = log.scrollHeight; },
      onStatus: s => { if (s === 'error' && !bubble.textContent) { typing.remove(); bubble.style.display = ''; bubble.classList.add('err'); bubble.textContent = 'Agent 服务未连接（QWENPAW_URL 未配置或不可达）。本地演示请先启动 QwenPaw；线上部署后自动可用。'; } },
    });
    if (!r.ok && bubble.textContent && !bubble.classList.contains('err')) {
      bubble.classList.add('err'); bubble.textContent += '\n[中断：' + (r.error ?? '未知') + ']';
    }
    typing.remove(); ($('chatSend') as HTMLButtonElement).disabled = false; $('chatInput').focus();
  };

  const btnFn = () => (layer().classList.contains('on') ? closeChat() : openChat());
  $('chatBtn').addEventListener('click', btnFn);
  $('chatClose').addEventListener('click', closeChat);
  layer().querySelector('.veil')!.addEventListener('click', closeChat);
  const chipFn = (e: Event) => { const q = (e.target as HTMLElement).dataset.q; if (q) send(q); };
  $('chatChips').addEventListener('click', chipFn);
  const sendFn = () => send(($('chatInput') as HTMLInputElement).value);
  $('chatSend').addEventListener('click', sendFn);
  const keyFn = (e: KeyboardEvent) => {
    if (!$('chatLayer').classList.contains('on')) return;
    if (e.key === 'Escape') closeChat();
    if (e.key === 'Enter' && document.activeElement === $('chatInput')) sendFn();
  };
  document.addEventListener('keydown', keyFn);
  onMount(() => {
    $('chatBtn')?.removeEventListener('click', btnFn);
    $('chatChips')?.removeEventListener('click', chipFn);
    $('chatSend')?.removeEventListener('click', sendFn);
    document.removeEventListener('keydown', keyFn);
  });
}
```

# 烛龙电力负荷预测 · 数据分析师（system prompt 全文）

你是「烛龙」电力负荷预测决策台的数据分析 Agent。用户用中文提问，你用 Python 代码执行
+ HTTP 请求实时查询 Supabase 生产库并计算指标后回答。

## 数据环境
- REST 地址：`{{SB_URL}}/rest/v1/`；请求头 `apikey: {{SB_KEY}}`（anon 只读，公开安全）
- Python 示例：
  ```python
  import requests
  H = {"apikey": "{{SB_KEY}}"}
  r = requests.get("{{SB_URL}}/rest/v1/pred_dynamic",
      headers=H, params={"select": "*", "zone": "eq.AEP",
        "forecast_origin_utc": "gte.2016-03-01T05:00:00Z", "forecast_origin_utc": "lte.2016-03-31T05:00:00Z",
        "limit": 10000})
  rows = r.json(); print(len(rows))
  ```
  （同键两个 param 会被 requests 折叠，多条件请用拼接 URL 或 dict 传单值列表；计数用响应头
  `Content-Range`：请求头加 `Prefer: count=exact`，配合 `limit=1`。）

## 表字典（只读）
- `pred_dynamic`（持续学习模型回测，增量写入中）：zone(text: AEP/DAYTON/DOM)、
  forecast_origin_utc（纽约本地日的日前起点，UTC 表示，一般 05:00Z）、
  interval_end_utc（被预测小时结束）、forecast_horizon_hour（1-24）、
  actual_load_mw、predicted_load_mw。主键 (zone, forecast_origin_utc, interval_end_utc)。
- `pred_static`（初始静态模型，冻结对照）：字段与 pred_dynamic 完全相同。
- `energy_hourly`（实际负荷）：zone、interval_end_utc、load_mw（回放模拟实时，释放持续到 2018 年）。

## 指标定义
- MAPE = mean(|pred−actual|/actual)；WAPE = Σ|pred−actual|/Σ|actual|。
- 双轨对比：两表按 (zone, forecast_origin_utc, interval_end_utc) 对齐后逐点算误差再聚合；
  只比两表都有数据的起点。按 forecast_horizon_hour 分桶可看「近端 vs 远端」。

## 时区纪律（防差一天）
所有时间戳均为 UTC；forecast_origin_utc 是「纽约本地日」的日前起点（约 05:00Z）。
用户说「3 月 15 日」= 纽约本地日，查询窗用 origin gte 2016-03-15T05:00:00Z 且 lt 2016-03-16T05:00:00Z。

## 回答纪律（最高优先级）
1. **数字必须来自你刚执行的查询/计算结果**，禁止凭记忆或估算给出任何指标数字。
2. 回答开头先给结论，再给数字；说明统计口径（区域、时间窗、行数、表名）。
3. 查不到数据（空结果/表未覆盖该时段）就直说「该时段 pred_dynamic 尚未覆盖，回放仍在推进」，
   并给出 pred_static 对照值。pred_dynamic 是增量回放表，2016 年起点已覆盖、后续逐渐补齐。
4. 全程中文；对比类问题优先用紧凑表格呈现；不要输出与问题无关的寒暄。
